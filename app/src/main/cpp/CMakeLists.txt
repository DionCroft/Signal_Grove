cmake_minimum_required(VERSION 3.31.6)
project(signal_grove_llm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJECT_PATH_MAX 250)
# Keep file names in native diagnostics without embedding paths from the build host.
add_compile_options(
    "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=signal-grove"
    "-ffile-prefix-map=${CMAKE_CURRENT_BINARY_DIR}=signal-grove-build")
if(ANDROID_ABI STREQUAL "armeabi-v7a" OR ANDROID_ABI STREQUAL "x86")
    # Keep the Hermes client installable on 32-bit devices. This shim explicitly
    # reports local inference as unavailable and never loads model weights.
    add_library(signal_grove_llm SHARED unsupported_local_llm.cpp)
    target_link_options(signal_grove_llm PRIVATE -Wl,-z,max-page-size=16384)
    return()
endif()
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(LLAMA_LLGUIDANCE OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
elseif(ANDROID_ABI STREQUAL "x86_64")
    set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
    set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
    foreach(flag AVX AVX2 AVX512 AVX512_VBMI AVX512_VNNI FMA F16C)
        set(GGML_${flag} OFF CACHE BOOL "" FORCE)
    endforeach()
else()
    message(FATAL_ERROR "Local inference supports ARM64 and x86_64 only")
endif()
include(FetchContent)
# A source archive, not an install script. All fetched/build files remain under the build directory.
FetchContent_Declare(llama_source
    URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/50518b64c6f2191a116d2760042f106c4b3387d7
    URL_HASH SHA256=df90368465d96bc7c5403afca4bafc8055608ad4499406f3870e784705644c08
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(llama_source)
add_library(signal_grove_llm SHARED signal_grove_llm.cpp)
option(SIGNAL_GROVE_CONTEXT_PROBE "Build the debug instrumentation context probe" OFF)
if(SIGNAL_GROVE_CONTEXT_PROBE)
    target_compile_definitions(signal_grove_llm PRIVATE SIGNAL_GROVE_CONTEXT_PROBE)
endif()
target_include_directories(signal_grove_llm PRIVATE ${llama_source_SOURCE_DIR}/common)
target_link_libraries(signal_grove_llm PRIVATE llama llama-common android log)
target_compile_options(signal_grove_llm PRIVATE -Wall -Wextra -Werror=return-type)
# Android 15+ can use 16-KiB pages, including devices other than the owner's phone.
target_link_options(signal_grove_llm PRIVATE -Wl,-z,max-page-size=16384)

# Explicit vocabulary-only device regression target. It is never packaged in an APK.
option(SIGNAL_GROVE_BUILD_NATIVE_TESTS "Build model-vocabulary output regression" OFF)
if(SIGNAL_GROVE_BUILD_NATIVE_TESTS)
    add_executable(signal_grove_chat_output_tests ../../test/cpp/native_chat_output_test.cpp)
    target_include_directories(signal_grove_chat_output_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${llama_source_SOURCE_DIR}/common)
    target_link_libraries(signal_grove_chat_output_tests PRIVATE llama llama-common android log)
    target_link_options(signal_grove_chat_output_tests PRIVATE -Wl,-z,max-page-size=16384)
endif()
