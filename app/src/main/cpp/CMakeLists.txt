cmake_minimum_required(VERSION 3.31.6)
project(signal_grove_llm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJECT_PATH_MAX 250)
if(ANDROID_ABI STREQUAL "armeabi-v7a" OR ANDROID_ABI STREQUAL "x86")
    # Keep the Hermes client installable on 32-bit devices. This shim explicitly
    # reports local inference as unavailable and never loads model weights.
    add_library(signal_grove_llm SHARED unsupported_local_llm.cpp)
    target_link_options(signal_grove_llm PRIVATE -Wl,-z,max-page-size=16384)
    return()
endif()
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(LLAMA_LLGUIDANCE OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
elseif(ANDROID_ABI STREQUAL "x86_64")
    set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
    set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
    foreach(flag AVX AVX2 AVX512 AVX512_VBMI AVX512_VNNI FMA F16C)
        set(GGML_${flag} OFF CACHE BOOL "" FORCE)
    endforeach()
else()
    message(FATAL_ERROR "Local inference supports ARM64 and x86_64 only")
endif()
include(FetchContent)
# A source archive, not an install script. All fetched/build files remain under the build directory.
FetchContent_Declare(llama_source
    URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/940a30ee3b5593e1a66fe0aadc3b731989022552
    URL_HASH SHA256=9cac27a1d867fec8eda2c5b348ac5f0b3b22e70fc4af46d82c93b64ab3f83167
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(llama_source)
add_library(signal_grove_llm SHARED signal_grove_llm.cpp)
target_include_directories(signal_grove_llm PRIVATE ${llama_source_SOURCE_DIR}/common)
target_link_libraries(signal_grove_llm PRIVATE llama llama-common android log)
target_compile_options(signal_grove_llm PRIVATE -Wall -Wextra -Werror=return-type)
# Android 15+ can use 16-KiB pages, including devices other than the owner's phone.
target_link_options(signal_grove_llm PRIVATE -Wl,-z,max-page-size=16384)
